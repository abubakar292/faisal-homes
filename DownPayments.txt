import React, { useEffect, useState, useMemo } from 'react';
import { useNavigate } from 'react-router-dom';
import { DollarSign, Search, Eye, CreditCard, X } from 'lucide-react';
import PageHeader from '../../components/ui/PageHeader';
import { getCollection, subscribeCollection } from '../../firebase/firestore';
import { formatCurrency } from '../../utils/formatters';
import type { Sale, Customer, Plot } from '../../types';

interface DownPaymentRow {
  saleId: string;
  saleNumber: string;
  customerId: string;        // firestore doc id
  customerDisplayId: string; // CUST-2026-001
  customerName: string;
  fatherName: string;
  phone: string;
  plotNumber: string;
  block: string;
  totalDownPayment: number;
  bookingPaid: number;
  remainingRequired: number;   // remainingDownPayment field
  receivedSoFar: number;       // downPaymentReceived field
  stillPending: number;        // remainingRequired - receivedSoFar
  status: 'complete' | 'partial' | 'pending';
}

const DownPayments: React.FC = () => {
  const navigate = useNavigate();
  const [sales, setSales]         = useState<Sale[]>([]);
  const [customers, setCustomers] = useState<Customer[]>([]);
  const [plots, setPlots]         = useState<Plot[]>([]);
  const [loading, setLoading]     = useState(true);
  const [search, setSearch]       = useState('');
  const [filterStatus, setFilterStatus] = useState<'all' | 'pending' | 'partial' | 'complete'>('all');

  useEffect(() => {
    const load = async () => {
      const [c, p] = await Promise.all([
        getCollection<Customer>('customers'),
        getCollection<Plot>('plots'),
      ]);
      setCustomers(c);
      setPlots(p);
      setLoading(false);
    };
    load();

    const unsub = subscribeCollection<Sale>('sales', (data) => {
      setSales(data.filter(s => s.status === 'active' && (s.remainingDownPayment || 0) > 0));
    });
    return () => unsub();
  }, []);

  const rows = useMemo<DownPaymentRow[]>(() => {
    const q = search.toLowerCase().trim();

    return sales.map(sale => {
      const customer = customers.find(c => c.id === sale.customerId);
      const plot     = plots.find(p => p.id === sale.plotId);
      if (!customer || !plot) return null;

      const remainingRequired = sale.remainingDownPayment || 0;
      const receivedSoFar     = sale.downPaymentReceived || 0;
      const stillPending      = Math.max(0, remainingRequired - receivedSoFar);

      let status: 'complete' | 'partial' | 'pending';
      if (stillPending === 0)       status = 'complete';
      else if (receivedSoFar > 0)   status = 'partial';
      else                          status = 'pending';

      return {
        saleId: sale.id,
        saleNumber: sale.saleNumber,
        customerId: customer.id,
        customerDisplayId: customer.customerId,
        customerName: customer.name,
        fatherName: customer.fatherName || '',
        phone: customer.phone,
        plotNumber: plot.plotNumber,
        block: plot.block,
        totalDownPayment: sale.downPaymentAmount || 0,
        bookingPaid: sale.bookingAmountPaid || 0,
        remainingRequired,
        receivedSoFar,
        stillPending,
        status,
      } as DownPaymentRow;
    }).filter((r): r is DownPaymentRow => {
      if (!r) return false;
      if (filterStatus !== 'all' && r.status !== filterStatus) return false;
      if (!q) return true;
      return (
        r.customerName.toLowerCase().includes(q) ||
        r.customerDisplayId.toLowerCase().includes(q) ||
        r.fatherName.toLowerCase().includes(q) ||
        r.phone.includes(q) ||
        r.plotNumber.toLowerCase().includes(q) ||
        r.block.toLowerCase().includes(q) ||
        r.saleNumber.toLowerCase().includes(q)
      );
    });
  }, [sales, customers, plots, search, filterStatus]);

  // Summary stats
  const totalPending  = rows.reduce((s, r) => s + r.stillPending, 0);
  const countComplete = rows.filter(r => r.status === 'complete').length;
  const countPartial  = rows.filter(r => r.status === 'partial').length;
  const countNotStart = rows.filter(r => r.status === 'pending').length;

  const statusBadge = (status: DownPaymentRow['status']) => {
    if (status === 'complete') return <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-bold" style={{ background: '#dcfce7', color: '#15803d' }}>✅ Fully Paid</span>;
    if (status === 'partial')  return <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-bold" style={{ background: '#fef3c7', color: '#b45309' }}>🔄 Partially Paid</span>;
    return                            <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-bold" style={{ background: '#fee2e2', color: '#b91c1c' }}>❌ Not Started</span>;
  };

  const filterBtn = (val: typeof filterStatus, label: string, color: string) => (
    <button
      onClick={() => setFilterStatus(val)}
      className="px-3 py-1.5 rounded-lg text-xs font-semibold transition-colors"
      style={filterStatus === val
        ? { background: color, color: 'white' }
        : { background: '#f1f5f9', color: '#64748b' }}
    >
      {label}
    </button>
  );

  return (
    <div className="space-y-6">
      <PageHeader
        title="Down Payment Management"
        subtitle="Track and collect remaining down payments for assigned plots"
        icon={<DollarSign size={22} />}
        actions={
          <button
            onClick={() => navigate('/payments/receive-down-payment')}
            className="btn btn-primary"
          >
            <CreditCard size={16} /> Receive Payment
          </button>
        }
      />

      {/* Summary Cards */}
      <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
        <div className="card text-center border-l-4" style={{ borderLeftColor: '#dc2626' }}>
          <p className="text-xs text-slate-500 mb-1">Total Pending Amount</p>
          <p className="text-xl font-extrabold" style={{ color: '#dc2626' }}>Rs. {formatCurrency(totalPending)}</p>
        </div>
        <div className="card text-center border-l-4" style={{ borderLeftColor: '#16a34a' }}>
          <p className="text-xs text-slate-500 mb-1">✅ Fully Paid</p>
          <p className="text-2xl font-extrabold" style={{ color: '#16a34a' }}>{countComplete}</p>
        </div>
        <div className="card text-center border-l-4" style={{ borderLeftColor: '#b45309' }}>
          <p className="text-xs text-slate-500 mb-1">🔄 Partially Paid</p>
          <p className="text-2xl font-extrabold" style={{ color: '#b45309' }}>{countPartial}</p>
        </div>
        <div className="card text-center border-l-4" style={{ borderLeftColor: '#b91c1c' }}>
          <p className="text-xs text-slate-500 mb-1">❌ Not Started</p>
          <p className="text-2xl font-extrabold" style={{ color: '#b91c1c' }}>{countNotStart}</p>
        </div>
      </div>

      {/* Filters */}
      <div className="card">
        <div className="flex flex-col sm:flex-row gap-3 mb-4">
          <div className="relative flex-1">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" size={16} />
            <input
              type="text"
              value={search}
              onChange={e => setSearch(e.target.value)}
              placeholder="Search by name, ID, father name, phone, plot..."
              className="input-field pl-9 pr-8"
            />
            {search && (
              <button onClick={() => setSearch('')} className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600">
                <X size={14} />
              </button>
            )}
          </div>
          <div className="flex gap-2">
            {filterBtn('all', 'All', '#1a3c6e')}
            {filterBtn('pending', '❌ Not Started', '#b91c1c')}
            {filterBtn('partial', '🔄 Partial', '#b45309')}
            {filterBtn('complete', '✅ Paid', '#16a34a')}
          </div>
        </div>

        {loading ? (
          <div className="flex justify-center py-12 text-slate-400">Loading...</div>
        ) : rows.length === 0 ? (
          <div className="flex flex-col items-center py-12 text-slate-400">
            <DollarSign size={40} className="mb-3 opacity-30" />
            <p>No down payment records found</p>
          </div>
        ) : (
          <div className="overflow-x-auto">
            <table className="w-full">
              <thead className="bg-slate-50">
                <tr>
                  {['Cust ID', 'Customer', 'Plot', 'Total DP', 'Booking Paid', 'Received', 'Still Pending', 'Status', 'Actions'].map(h => (
                    <th key={h} className="table-th text-left">{h}</th>
                  ))}
                </tr>
              </thead>
              <tbody>
                {rows.map(row => (
                  <tr key={row.saleId} className="table-row">
                    <td className="table-td">
                      <span className="font-mono text-xs px-2 py-0.5 rounded" style={{ background: '#eff6ff', color: '#1d4ed8' }}>
                        {row.customerDisplayId}
                      </span>
                    </td>
                    <td className="table-td">
                      <p className="font-semibold text-slate-800">{row.customerName}</p>
                      <p className="text-xs text-slate-400">{row.phone}</p>
                    </td>
                    <td className="table-td">
                      <p className="font-semibold">{row.plotNumber}</p>
                      <p className="text-xs text-slate-400">Block {row.block}</p>
                    </td>
                    <td className="table-td font-semibold">Rs. {formatCurrency(row.totalDownPayment)}</td>
                    <td className="table-td text-slate-500">Rs. {formatCurrency(row.bookingPaid)}</td>
                    <td className="table-td" style={{ color: '#16a34a', fontWeight: 600 }}>Rs. {formatCurrency(row.receivedSoFar)}</td>
                    <td className="table-td font-bold" style={{ color: row.stillPending > 0 ? '#b91c1c' : '#16a34a' }}>
                      Rs. {formatCurrency(row.stillPending)}
                    </td>
                    <td className="table-td">{statusBadge(row.status)}</td>
                    <td className="table-td">
                      <div className="flex gap-1">
                        <button
                          onClick={() => navigate(`/payments/down-payment-ledger?saleId=${row.saleId}`)}
                          className="flex items-center gap-1 px-2 py-1 rounded-lg text-xs font-medium hover:bg-blue-50 transition-colors"
                          style={{ color: '#1d4ed8' }}
                          title="View Ledger"
                        >
                          <Eye size={13} /> Ledger
                        </button>
                        {row.stillPending > 0 && (
                          <button
                            onClick={() => navigate(`/payments/receive-down-payment?saleId=${row.saleId}`)}
                            className="flex items-center gap-1 px-2 py-1 rounded-lg text-xs font-medium hover:bg-green-50 transition-colors"
                            style={{ color: '#16a34a' }}
                            title="Receive Payment"
                          >
                            <CreditCard size={13} /> Receive
                          </button>
                        )}
                      </div>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>
    </div>
  );
};

export default DownPayments;
